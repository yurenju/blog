```ts
// src/lib/seo.ts
import { HREFLANG, type Locale } from './i18n';
import type { PostMeta } from './posts';

/**
 * Default OG image when a page has no specific cover.
 * Site logo lives at root /logo.jpg (Phase 0+).
 */
export const SITE_LOGO = {
  url: 'https://yurenju.blog/logo.jpg',
  width: 100,
  height: 100,
};

/**
 * Resolve a path or URL to an absolute URL using the configured site origin.
 * Pass-through if already absolute (http/https).
 */
export function absoluteUrl(pathOrUrl: string, site: URL): string {
  if (/^https?:/.test(pathOrUrl)) return pathOrUrl;
  return new URL(pathOrUrl, site).href;
}

/**
 * Build the JSON-LD `Article` schema object for a post.
 * Caller is responsible for serializing via JSON.stringify (and escaping `</`).
 */
export function buildArticleSchema(
  post: PostMeta,
  locale: Locale,
  ogImageUrl: string,
): object {
  return {
    '@context': 'https://schema.org',
    '@type': 'Article',
    headline: post.title,
    description: post.description,
    inLanguage: HREFLANG[locale],
    datePublished: post.date.toISOString(),
    image: ogImageUrl,
    author: {
      '@type': 'Person',
      name: 'Yuren Ju',
      url: 'https://yurenju.blog',
    },
  };
}
```

## Task 4：修改 `PostLayout.astro` — 算 ogImage、產 JSON-LD、傳 SEO props

**Files:**
- Modify: `src/layouts/PostLayout.astro`

`PostLayout`：
- 計算 `ogImage`：有 `post.cover` 就用 `cover.src + width + height`、否則 fallback `SITE_LOGO`
- 把 cover URL 用 `absoluteUrl()` 轉絕對 URL
- 傳 `canonical={Astro.url}`、`ogType="article"`、`ogImage`、`ogLocale={HTML_LANG[locale]}` 給 BaseLayout
- 在 `<Fragment slot="head">` 內加 `<script type="application/ld+json">`（與既有 hreflang 共存）

- [ ] **Step 1: 替換 `src/layouts/PostLayout.astro` 整檔**

```astro
---
import BaseLayout from './BaseLayout.astro';
import Header from '../components/Header.astro';
import Footer from '../components/Footer.astro';
import PostMeta from '../components/PostMeta.astro';
import ArticleLanguageIndicator from '../components/ArticleLanguageIndicator.astro';
import type { PostMeta as PostMetaType } from '../lib/posts';
import { HTML_LANG, HREFLANG, type Locale } from '../lib/i18n';
import { SITE_LOGO, absoluteUrl, buildArticleSchema } from '../lib/seo';

interface Props {
  locale: Locale;
  post: PostMetaType;
}

const { locale, post } = Astro.props;
const pageTitle = `${post.title} · Yuren's Blog`;
const siteUrl = 'https://yurenju.blog';

// Compute ogImage: prefer post cover (hashed _astro URL), else site logo.
const ogImage = post.cover
  ? {
      url: absoluteUrl(post.cover.src, Astro.site!),
      width: post.cover.width,
      height: post.cover.height,
    }
  : SITE_LOGO;

const articleSchema = buildArticleSchema(post, locale, ogImage.url);
const articleSchemaJson = JSON.stringify(articleSchema).replaceAll('</', '<\\/');
---
<BaseLayout
  title={pageTitle}
  description={post.description}
  lang={HTML_LANG[locale]}
  canonical={Astro.url}
  ogType="article"
  ogImage={ogImage}
  ogLocale={HTML_LANG[locale]}
>
  <Fragment slot="head">
    {post.availableLocales.map((l) => (
      <link
        rel="alternate"
        hreflang={HREFLANG[l]}
        href={`${siteUrl}/${l}/posts/${post.slug}`}
      />
    ))}
    <link rel="alternate" hreflang="x-default" href={`${siteUrl}/zh/posts/${post.slug}`} />
    <script type="application/ld+json" set:html={articleSchemaJson} />
  </Fragment>
  <Header
    locale={locale}
    pathname={Astro.url.pathname}
    isPostPage={true}
    slug={post.slug}
    availableLocales={post.availableLocales}
  />
  <main class="container">
    <article>
      <header class="post-head">
        <h1>{post.title}</h1>
        <PostMeta date={post.date} category={post.category} />
        <ArticleLanguageIndicator
          currentLocale={locale}
          availableLocales={post.availableLocales}
          slug={post.slug}
        />
      </header>
      <div class="prose">
        <slot />
      </div>
    </article>
  </main>
  <Footer />
</BaseLayout>

<style>
  .post-head {
    margin-bottom: 2rem;
    padding-bottom: 1.5rem;
    border-bottom: 1px solid #eee;
  }
  .post-head h1 {
    font-size: 2.25rem;
    margin-bottom: 0.75rem;
  }
</style>
```
